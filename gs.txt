// ฟังก์ชันสำหรับตรวจเช็คและสร้างชีตอัตโนมัติหากยังไม่มี

  // ตรวจสอบและสร้างชีต Products
  const pSheet = ss.getSheetByName("Products");

    sheet.getRange("1:1").setFontWeight("bold").setBackground("#d9ead3");
  } else {
    const lastCol = pSheet.getLastColumn();
    if (lastCol < 11) pSheet.getRange(1, 11).setValue("Stock_Qty");
    if (lastCol < 12) pSheet.getRange(1, 12).setValue("Product_Group");
    if (lastCol < 13) pSheet.getRange(1, 13).setValue("Supplier");
    if (lastCol < 14) pSheet.getRange(1, 14).setValue("Storage_Area");

  if (!ss.getSheetByName("Machines")) {

// ฟังก์ชันดึงข้อมูลทั้งหมดส่งไปให้ Frontend ในรูปแบบ JSON
function getAppData() {
  initDatabase();
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  
  // ดึงข้อมูล Products
  const prodSheet = ss.getSheetByName("Products");
  const prodData = prodSheet.getDataRange().getValues();
  const products = [];
  if (prodData.length > 1) {
    for (let i = 1; i < prodData.length; i++) {
      let r = prodData[i];
      products.push({
        id: r[0], name: r[1], unit: r[2], cost: r[3],
        price_a: r[4], price_b: r[5], price_c: r[6],
        category: r[7], note: r[8], image_url: r[9],
        stock_qty: (r[10] !== undefined && r[10] !== "") ? parseFloat(r[10]) : 0,
        group: r[11] || "",
        supplier: r[12] || "",
        storage: r[13] || ""
      });
    }
  }

  // ดึงข้อมูล Machines
  const machSheet = ss.getSheetByName("Machines");
  const machData = machSheet.getDataRange().getValues();
  const machines = [];
  if (machData.length > 1) {
    for (let i = 1; i < machData.length; i++) {
      machines.push({ 
        id: machData[i][0], 
        name: machData[i][1], 
        image_url: machData[i][2] || '',
        cost: machData[i][3] || 0,
        price_a: machData[i][4] || 0,
        price_b: machData[i][5] || 0,
        price_c: machData[i][6] || 0,
        note: machData[i][7] || '',
        group: machData[i][8] || '',
        supplier: machData[i][9] || '',
        storage: machData[i][10] || ''
      });
    }
  }

  // ดึงข้อมูล Mapping จาก JSON ใน Google Drive
  let mappings = getMappingsFromJson();
  const machineIds = new Set(machines.map(m => String(m.id).trim()));
  const productIds = new Set(products.map(p => String(p.id).trim()));
  const initialMappingsLength = mappings.length;

  mappings = mappings.filter(m => {
    return machineIds.has(String(m.machine_id).trim()) && 
           productIds.has(String(m.product_id).trim());
  });
  if (mappings.length < initialMappingsLength) {
    try {
      saveMappingsToJson(mappings);
    } catch (e) {
      Logger.log("ไม่สามารถเซฟข้อมูลการจับคู่ที่กรองใหม่ได้: " + e.toString());
    }
  }

  // ดึงข้อมูล Settings
  const settingsSheet = ss.getSheetByName("Settings");
  const settings = { isShowPriceBForGuest: false, isShowPriceCForGuest: false };
  if (settingsSheet) {
    const settingsData = settingsSheet.getDataRange().getValues();
    if (settingsData.length > 1) {
      for (let i = 1; i < settingsData.length; i++) {
        let key = settingsData[i][0];
        let val = settingsData[i][1];
        if (key === "isShowPriceBForGuest") settings.isShowPriceBForGuest = (String(val).toLowerCase() === "true");
        if (key === "isShowPriceCForGuest") settings.isShowPriceCForGuest = (String(val).toLowerCase() === "true");
      }
    }
  }

  // ดึงข้อมูล Manuals
  const manSheet = ss.getSheetByName("Manuals");
  const manuals = [];
  if (manSheet) {
    const manData = manSheet.getDataRange().getValues();
    if (manData.length > 1) {
      for (let i = 1; i < manData.length; i++) {
        manuals.push({
          id: manData[i][0],
          title: manData[i][1],
          description: manData[i][2] || '',
          file_url: manData[i][3] || '',
          file_type: manData[i][4] || '',
          uploaded_at: manData[i][5] || ''
        });
      }
    }
  // ดึงข้อมูลผู้อนุมัติที่มีสิทธิ์ (canApprove)
  let approvers = [];
  try {
    const allUsers = getUsersJSON();
    approvers = allUsers.filter(u => !!u.canApprove).map(u => ({
      fullName: u.fullName || "",
      email: u.email || "",
      department: u.department || "",
      role: u.role || "User"
    }));
  } catch (e) {
    Logger.log("ไม่สามารถดึงข้อมูลผู้อนุมัติได้: " + e.toString());
  }

  return { products: products, machines: machines, mappings: mappings, settings: settings, manuals: manuals, lots: getLotsFromJson(), approvers: approvers };
}

// ฟังก์ชันอัปโหลดภาพและสร้างลิงก์ lh3
function uploadImageAndGetLh3Url(base64Data, fileName) {
  try {
    if (!base64Data) return "";
    const base64 = base64Data.split(',')[1] || base64Data;
    const decoded = Utilities.base64Decode(base64);
    const blob = Utilities.newBlob(decoded, "image/jpeg", fileName); 
    
    const folder = DriveApp.getFolderById(DRIVE_FOLDER_ID);
    const file = folder.createFile(blob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    
    return 'https://lh3.googleusercontent.com/d/' + file.getId();
  } catch (e) {
    throw new Error("อัปโหลดรูปภาพล้มเหลว: " + e.message);
  }
}

// ฟังก์ชันคำนวณราคา 3 ระดับตามเงื่อนไข (ใช้กับอะไหล่)
function calculatePrices(cost) {
  const costNum = parseFloat(cost) || 0;
  return {
    price_a: costNum * 2.1,
    price_b: costNum * 1.7,
    price_c: costNum * 1.3
  };
}

// 1. เพิ่มข้อมูล Product
function addProduct(data) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = ss.getSheetByName("Products");
  
  let imageUrl = "";
  if (data.imageBase64) {
    const fileName = "IMG_" + data.id + "_" + new Date().getTime() + ".jpg";
    imageUrl = uploadImageAndGetLh3Url(data.imageBase64, fileName);
  }

  const autoPrices = calculatePrices(data.cost);
  const pA = parseFloat(data.price_a) > 0 ? parseFloat(data.price_a) : autoPrices.price_a;
  const pB = parseFloat(data.price_b) > 0 ? parseFloat(data.price_b) : autoPrices.price_b;
  const pC = parseFloat(data.price_c) > 0 ? parseFloat(data.price_c) : autoPrices.price_c;
  
  sheet.appendRow([
    data.id, data.name, data.unit, data.cost,
    pA, pB, pC,
    data.category, data.note, imageUrl,
    (data.stock_qty !== undefined && data.stock_qty !== "") ? parseFloat(data.stock_qty) : 0,
    data.group || "", data.supplier || "", data.storage || ""
  ]);
  return { image_url: imageUrl };
}

// 2. แก้ไขข้อมูล Product
function editProduct(data) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = ss.getSheetByName("Products");
  const dataRange = sheet.getDataRange();
  const values = dataRange.getValues();
  
  let rowIndex = -1;
  for (let i = 1; i < values.length; i++) {
    if (values[i][0] == data.id) {
      rowIndex = i + 1;
      break;
    }
  }

  if (rowIndex > -1) {
    let imageUrl = values[rowIndex-1][9];
    if (data.imageBase64) {
      const fileName = "IMG_" + data.id + "_" + new Date().getTime() + ".jpg";
      imageUrl = uploadImageAndGetLh3Url(data.imageBase64, fileName);
    }

    const autoPrices = calculatePrices(data.cost);
    const pA = parseFloat(data.price_a) > 0 ? parseFloat(data.price_a) : autoPrices.price_a;
    const pB = parseFloat(data.price_b) > 0 ? parseFloat(data.price_b) : autoPrices.price_b;
    const pC = parseFloat(data.price_c) > 0 ? parseFloat(data.price_c) : autoPrices.price_c;

    let stockQty = (data.stock_qty !== undefined && data.stock_qty !== "") ? parseFloat(data.stock_qty) : parseFloat(values[rowIndex-1][10] || 0);

    sheet.getRange(rowIndex, 2, 1, 13).setValues([[
      data.name, data.unit, data.cost,
      pA, pB, pC,
      data.category, data.note, imageUrl,
      stockQty,
      data.group || "", data.supplier || "", data.storage || ""
    ]]);
  } else {
    throw new Error("ไม่พบรหัสสินค้าที่ต้องการแก้ไข");
  }
}

// 3. ลบข้อมูล Product
function deleteProduct(productId) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = ss.getSheetByName("Products");
  const values = sheet.getDataRange().getValues();
  
  const pid = String(productId).trim();
  for (let i = 1; i < values.length; i++) {
    if (String(values[i][0]).trim() === pid) {
      sheet.deleteRow(i + 1);
      break;
    }
  }
  
  let mappings = getMappingsFromJson();
  let newMappings = mappings.filter(m => String(m.product_id).trim() !== pid);
  if (newMappings.length !== mappings.length) {
    saveMappingsToJson(newMappings);
  }
}

      let imageUrl = values[i][2] || '';

  for (let i = 1; i < values.length; i++) {
    if (String(values[i][0]).trim() === mid) {
      sheet.deleteRow(i + 1);
      break;
    }
  }
  
  let mappings = getMappingsFromJson();
  let newMappings = mappings.filter(m => String(m.machine_id).trim() !== mid);
  if (newMappings.length !== mappings.length) {
    saveMappingsToJson(newMappings);
  }
}

// 7. เพิ่ม Mapping
function addMapping(data) {
  let productIds = Array.isArray(data.product_ids) ? data.product_ids : [data.product_id];
  let machineId = String(data.machine_id);
  
  let mappings = getMappingsFromJson();
  let isModified = false;
  for (let p = 0; p < productIds.length; p++) {
    let pid = String(productIds[p]);
    if(!pid) continue;
    let isDuplicate = mappings.some(m => String(m.product_id) === pid && String(m.machine_id) === machineId);
    if (!isDuplicate) {
      mappings.push({ product_id: pid, machine_id: machineId });
      isModified = true;
    }
  }
  
  if (isModified) {
    saveMappingsToJson(mappings);
  } else {
    throw new Error("รายการอะไหล่ที่เลือก ถูกจับคู่กับเครื่องจักรนี้อยู่แล้วทั้งหมด");
  }
}

// 8. ลบ Mapping
function deleteMapping(data) {
  let pid = String(data.product_id);
  let mid = String(data.machine_id);
  
  let mappings = getMappingsFromJson();
  let initialLength = mappings.length;
  mappings = mappings.filter(m => !(String(m.product_id) === pid && String(m.machine_id) === mid));
  if (mappings.length < initialLength) {
    saveMappingsToJson(mappings);
  } else {
    throw new Error("ไม่พบรายการจับคู่ที่ต้องการลบ");
  }
}

function migrateTransactionsToJson() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const txSheet = ss.getSheetByName("Transactions");
  const txdSheet = ss.getSheetByName("Transaction_Details");
  
  if (!txSheet || !txdSheet) return [];
  
  const txValues = txSheet.getDataRange().getValues();
  const txdValues = txdSheet.getDataRange().getValues();
  const detailsMap = {};
  
  if (txdValues.length > 1) {
    for (let i = 1; i < txdValues.length; i++) {
      const r = txdValues[i];
      const txId = String(r[1]).trim();
      if (!detailsMap[txId]) detailsMap[txId] = [];
      detailsMap[txId].push({
        detail_id: r[0], product_id: r[2], qty: r[3], price: r[4], subtotal: r[5]
      });
    }
  }
  
  const transactions = [];
  if (txValues.length > 1) {
    for (let i = 1; i < txValues.length; i++) {
      const r = txValues[i];
      let dateVal = r[1];
      if (dateVal instanceof Date) {
        dateVal = Utilities.formatDate(dateVal, Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss");
      }
      transactions.push({
        id: String(r[0]).trim(),
        date: dateVal,
        requester: r[2],
        department: r[3],
        machine_id: r[4],
        total_price: r[5],
        note: r[6],
        status: r[7],
        items: detailsMap[String(r[0]).trim()] || []
      });
    }
  }
  
  saveTransactionsToJson(transactions);
  return transactions;
}

// 9. บันทึกการทำรายการเบิกอะไหล่ (POS Checkout) พร้อมตัดสต็อกตาม FIFO
function checkoutOrder(payload) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const prodSheet = ss.getSheetByName("Products");
  
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(30000);
  } catch (e) {
    throw new Error("ระบบไม่ว่างชั่วคราว กรุณาลองใหม่อีกครั้ง");
  }
  
  try {
    const prodValues = prodSheet.getDataRange().getValues();
    const cart = payload.cart; 
    let lots = getLotsFromJson();
    
    const prodRowMap = {};
    for (let i = 1; i < prodValues.length; i++) {
      const pid = String(prodValues[i][0]).trim();
      prodRowMap[pid] = {
        rowIndex: i + 1,
        name: prodValues[i][1],
        stock: parseFloat(prodValues[i][10]) || 0,
        cost: parseFloat(prodValues[i][3]) || 0,
        price_a: parseFloat(prodValues[i][4]) || 0,
        price_b: parseFloat(prodValues[i][5]) || 0,
        price_c: parseFloat(prodValues[i][6]) || 0
      };
    }
    
    // ตรวจเช็คสต็อกคงเหลือ
    for (let k = 0; k < cart.length; k++) {
      const item = cart[k];
      const prodId = String(item.id).trim();
      const product = prodRowMap[prodId];
      if (!product) throw new Error("ไม่พบอะไหล่รหัส " + prodId + " ในระบบ");
      if (product.stock < item.qty) {
        throw new Error("สต็อกไม่เพียงพอ: อะไหล่ " + product.name + " (" + prodId + ") มีคงเหลือ " + product.stock + " ชิ้น แต่พยายามเบิก " + item.qty + " ชิ้น");
      }
      
      // ตรวจสอบข้อมูล Lot ว่าพอเพียงหรือไม่ ถ้า Lot ยังน้อยกว่า stock_qty ให้สร้าง Lot สำรองให้อัตโนมัติ
      const prodLots = lots.filter(l => String(l.product_id).trim() === prodId && l.remaining_qty > 0);
      const totalLotQty = prodLots.reduce((sum, l) => sum + l.remaining_qty, 0);
      if (totalLotQty < product.stock) {
        const diff = product.stock - totalLotQty;
        lots.push({
          lot_id: "LOT-SUPP-" + prodId + "-" + Date.now(),
          product_id: prodId,
          cost: product.cost,
          price_a: product.price_a,
          price_b: product.price_b,
          price_c: product.price_c,
          initial_qty: diff,
          remaining_qty: diff,
          created_at: Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss"),
          note: "Lot สำรองคงเหลือ"
        });
      }
    }
    
    // ตัดสต็อกตาม FIFO (First-In, First-Out)
    const checkoutItems = [];
    
    for (let k = 0; k < cart.length; k++) {
      const item = cart[k];
      const prodId = String(item.id).trim();
      let neededQty = item.qty;
      
      // ดึง Lot ของสินค้าตัวนี้ที่ยังเหลืออยู่ เรียงจากสร้างก่อนไปหลัง (FIFO)
      const availableLots = lots.filter(l => String(l.product_id).trim() === prodId && l.remaining_qty > 0)
                                .sort((a, b) => new Date(a.created_at || 0) - new Date(b.created_at || 0));
      
      for (let l = 0; l < availableLots.length && neededQty > 0; l++) {
        const currentLot = availableLots[l];
        const takeQty = Math.min(neededQty, currentLot.remaining_qty);
        
        currentLot.remaining_qty -= takeQty;
        neededQty -= takeQty;
        
        // กำหนดราคาเบิกต่อหน่วยตาม Lot (ถ้ากำหนดระดับราคา A/B/C)
        let lotPrice = item.price;
        if (item.priceLevel === 'A' && currentLot.price_a) lotPrice = currentLot.price_a;
        else if (item.priceLevel === 'B' && currentLot.price_b) lotPrice = currentLot.price_b;
        else if (item.priceLevel === 'C' && currentLot.price_c) lotPrice = currentLot.price_c;
        
        checkoutItems.push({
          detail_id: "",
          product_id: prodId,
          lot_id: currentLot.lot_id,
          qty: takeQty,
          unit_cost: currentLot.cost || 0,
          price: lotPrice,
          subtotal: takeQty * lotPrice
        });
      }
      
      // หักสต็อกในชีต Products
      const product = prodRowMap[prodId];
      const newStock = product.stock - item.qty;
      prodSheet.getRange(product.rowIndex, 11).setValue(newStock);
      product.stock = newStock;
    }
    
    saveLotsToJson(lots);
    
    const now = new Date();
    const dateStr = Utilities.formatDate(now, Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss");
    const datePrefix = Utilities.formatDate(now, Session.getScriptTimeZone(), "yyyyMMdd");
    
    let transactions = getTransactionsFromJson();
    let counter = 1;
    
    if (transactions.length > 0) {
      const lastTx = transactions[transactions.length - 1];
      if (String(lastTx.id).indexOf("QCM-" + datePrefix) === 0) {
        const parts = String(lastTx.id).split("-");
        const lastNum = parseInt(parts[2], 10);
        if (!isNaN(lastNum)) counter = lastNum + 1;
      }
    }
    const txId = "QCM-" + datePrefix + "-" + String(counter).padStart(4, '0');
    
    let calcTotalPrice = 0;
    checkoutItems.forEach((it, idx) => {
      it.detail_id = txId + "-" + (idx + 1);
      calcTotalPrice += it.subtotal;
    });
    
    const laborCost = parseFloat(payload.labor_cost) || 0;
    const partsTotal = parseFloat(payload.parts_total) || calcTotalPrice;
    const grandTotal = parseFloat(payload.total_price) || (calcTotalPrice + laborCost);

    const newTransaction = {
      id: txId,
      date: dateStr,
      requester: payload.requester || "",
      department: payload.department || "",
      approver: payload.approver || "",
      approver_email: payload.approver_email || "",
      customer_type: payload.customer_type || "",
      price_tier: payload.price_tier || "",
      machine_model: payload.machine_model || "",
      repair_level: payload.repair_level || "",
      labor_cost: laborCost,
      parts_total: partsTotal,
      machine_id: payload.machine_id,
      serial_number: payload.serial_number,
      total_price: grandTotal,
      note: payload.note,
      job_details: payload.job_details || "",
      status: "Success",
      approval_status: payload.approval_status || "Pending",
      items: checkoutItems
    };
    
    transactions.push(newTransaction);
    saveTransactionsToJson(transactions);
    
    return { transaction_id: txId, items: checkoutItems };
    
  } finally {
    lock.releaseLock();
  }
}

// 10. ดึงข้อมูลประวัติใบเบิกอะไหล่ทั้งหมด
function getTransactions() {
  let transactions = getTransactionsFromJson();
  return transactions.reverse();
}

// 11. ยกเลิกใบเบิกสินค้า (คืนสต็อกกลับเข้า Products และ Lot เดิม)
function cancelTransaction(payload) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const prodSheet = ss.getSheetByName("Products");
  
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(30000);
  } catch (e) {
    throw new Error("ระบบไม่ว่างชั่วคราว กรุณาลองใหม่อีกครั้ง");
  }
  
  try {
    const txId = String(payload.transaction_id).trim();
    let transactions = getTransactionsFromJson();
    
    const txIndex = transactions.findIndex(t => t.id === txId);
    if (txIndex === -1) throw new Error("ไม่พบรหัสใบเบิก " + txId);
    
    const targetTx = transactions[txIndex];
    if (targetTx.status === "Cancelled") throw new Error("ใบเบิกนี้ถูกยกเลิกไปแล้ว");
    
    const prodValues = prodSheet.getDataRange().getValues();
    const prodRowMap = {};
    for (let i = 1; i < prodValues.length; i++) {
      prodRowMap[String(prodValues[i][0]).trim()] = {
        rowIndex: i + 1,
        stock: parseFloat(prodValues[i][10]) || 0
      };
    }
    
    let lots = getLotsFromJson();
    
    for (let k = 0; k < targetTx.items.length; k++) {
      const item = targetTx.items[k];
      const prodId = String(item.product_id).trim();
      const product = prodRowMap[prodId];
      if (product) {
        prodSheet.getRange(product.rowIndex, 11).setValue(product.stock + item.qty);
        product.stock += item.qty;
      }
      
      if (item.lot_id) {
        const targetLot = lots.find(l => l.lot_id === item.lot_id);
        if (targetLot) {
          targetLot.remaining_qty += item.qty;
        }
      }
    }
    
    saveLotsToJson(lots);
    
    transactions[txIndex].status = "Cancelled";
    saveTransactionsToJson(transactions);
    
    return { status: "success", message: "ยกเลิกใบเบิกสำเร็จ คืนสต็อกอะไหล่เข้าคลังเรียบร้อยแล้ว" };
  } finally {
    lock.releaseLock();
  }
}

// 11.05 อนุมัติ หรือ ไม่อนุมัติ หรือ แก้ไขรายการอะไหล่ในเอกสารใบขอเบิก
function updateApprovalStatus(payload) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(30000);
  } catch (e) {
    throw new Error("ระบบไม่ว่างชั่วคราว กรุณาลองใหม่อีกครั้ง");
  }
  
  try {
    const txId = String(payload.transaction_id).trim();
    const newStatus = payload.approval_status; // 'Approved', 'Rejected', or 'Pending'
    const note = payload.approval_note || "";
    const approverName = payload.approved_by || "";
    const approverEmail = payload.approved_by_email || "";
    
    let transactions = getTransactionsFromJson();
    const txIndex = transactions.findIndex(t => t.id === txId);
    if (txIndex === -1) throw new Error("ไม่พบรหัสใบเบิก " + txId);
    
    const targetTx = transactions[txIndex];
    if (targetTx.approval_status === "Approved" || targetTx.approval_status === "Rejected") {
      throw new Error("เอกสารนี้ได้รับการพิจารณาไปแล้ว (" + targetTx.approval_status + ")");
    }
    
    const now = new Date();
    const dateStr = Utilities.formatDate(now, Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss");
    
    const hasUpdatedItems = Array.isArray(payload.updated_items) && payload.updated_items.length > 0;

    if (newStatus === "Rejected") {
      targetTx.approval_status = "Rejected";
      targetTx.approval_date = dateStr;
      targetTx.approval_by = approverName;
      targetTx.approval_by_email = approverEmail;
      targetTx.approval_note = note;
      targetTx.status = "Cancelled";
      
      const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
      const prodSheet = ss.getSheetByName("Products");
      const prodValues = prodSheet.getDataRange().getValues();
      const prodRowMap = {};
      for (let i = 1; i < prodValues.length; i++) {
        prodRowMap[String(prodValues[i][0]).trim()] = {
          rowIndex: i + 1,
          stock: parseFloat(prodValues[i][10]) || 0
        };
      }
      
      let lots = getLotsFromJson();
      
      if (Array.isArray(targetTx.items)) {
        for (let k = 0; k < targetTx.items.length; k++) {
          const item = targetTx.items[k];
          const prodId = String(item.product_id).trim();
          const product = prodRowMap[prodId];
          if (product) {
            prodSheet.getRange(product.rowIndex, 11).setValue(product.stock + item.qty);
            product.stock += item.qty;
          }
          
          if (item.lot_id) {
            const targetLot = lots.find(l => l.lot_id === item.lot_id);
            if (targetLot) {
              targetLot.remaining_qty += item.qty;
            }
          }
        }
      }
      
      saveLotsToJson(lots);
    } else {
      // 'Approved' or 'Pending'
      if (hasUpdatedItems) {
        const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
        const prodSheet = ss.getSheetByName("Products");
        const prodValues = prodSheet.getDataRange().getValues();
        const prodRowMap = {};
        for (let i = 1; i < prodValues.length; i++) {
          const pid = String(prodValues[i][0]).trim();
          prodRowMap[pid] = {
            rowIndex: i + 1,
            name: prodValues[i][1],
            stock: parseFloat(prodValues[i][10]) || 0,
            cost: parseFloat(prodValues[i][3]) || 0,
            price_a: parseFloat(prodValues[i][4]) || 0,
            price_b: parseFloat(prodValues[i][5]) || 0,
            price_c: parseFloat(prodValues[i][6]) || 0
          };
        }
        
        let lots = getLotsFromJson();
        
        // 1. คืนสต็อกและล็อตเดิมก่อนคำนวณใหม่
        if (Array.isArray(targetTx.items)) {
          for (let k = 0; k < targetTx.items.length; k++) {
            const oldItem = targetTx.items[k];
            const oldPid = String(oldItem.product_id).trim();
            const oldProd = prodRowMap[oldPid];
            if (oldProd) {
              oldProd.stock += (Number(oldItem.qty) || 0);
            }
            if (oldItem.lot_id) {
              const targetLot = lots.find(l => l.lot_id === oldItem.lot_id);
              if (targetLot) {
                targetLot.remaining_qty += (Number(oldItem.qty) || 0);
              }
            }
          }
        }
        
        // 2. ตรวจสอบสต็อกสำหรับรายการใหม่ทั้งหมด
        const productNeededTotals = {};
        for (let k = 0; k < payload.updated_items.length; k++) {
          const item = payload.updated_items[k];
          const prodId = String(item.product_id || item.id).trim();
          const q = Number(item.qty) || 0;
          if (q <= 0) throw new Error("จำนวนอะไหล่ต้องมากกว่า 0");
          productNeededTotals[prodId] = (productNeededTotals[prodId] || 0) + q;
        }

        for (const prodId in productNeededTotals) {
          const product = prodRowMap[prodId];
          if (!product) throw new Error("ไม่พบรหัสสินค้า " + prodId + " ในระบบ");
          const totalReq = productNeededTotals[prodId];
          if (product.stock < totalReq) {
            throw new Error("สต็อกไม่เพียงพอ: อะไหล่ " + product.name + " (" + prodId + ") มีคงเหลือ " + product.stock + " ชิ้น แต่ต้องการเบิก " + totalReq + " ชิ้น");
          }
          
          // ตรวจสอบ Lot สำรอง
          const prodLots = lots.filter(l => String(l.product_id).trim() === prodId && l.remaining_qty > 0);
          const totalLotQty = prodLots.reduce((sum, l) => sum + l.remaining_qty, 0);
          if (totalLotQty < product.stock) {
            const diff = product.stock - totalLotQty;
            lots.push({
              lot_id: "LOT-SUPP-" + prodId + "-" + Date.now() + "-" + Math.floor(Math.random() * 1000),
              product_id: prodId,
              cost: product.cost,
              price_a: product.price_a,
              price_b: product.price_b,
              price_c: product.price_c,
              initial_qty: diff,
              remaining_qty: diff,
              created_at: Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss"),
              note: "Lot สำรองคงเหลือ (ปรับปรุงรายการอนุมัติ)"
            });
          }
        }
        
        // 3. ตัดสต็อกตาม FIFO สำหรับรายการใหม่
        const checkoutItems = [];
        for (let k = 0; k < payload.updated_items.length; k++) {
          const item = payload.updated_items[k];
          const prodId = String(item.product_id || item.id).trim();
          const product = prodRowMap[prodId];
          let neededQty = Number(item.qty) || 0;
          
          const availableLots = lots.filter(l => String(l.product_id).trim() === prodId && l.remaining_qty > 0)
                                    .sort((a, b) => new Date(a.created_at || 0) - new Date(b.created_at || 0));
          
          for (let l = 0; l < availableLots.length && neededQty > 0; l++) {
            const currentLot = availableLots[l];
            const takeQty = Math.min(neededQty, currentLot.remaining_qty);
            currentLot.remaining_qty -= takeQty;
            neededQty -= takeQty;
            
            const lotPrice = Number(item.price) || 0;
            checkoutItems.push({
              detail_id: "",
              product_id: prodId,
              lot_id: currentLot.lot_id,
              qty: takeQty,
              unit_cost: currentLot.cost || 0,
              price: lotPrice,
              subtotal: takeQty * lotPrice
            });
          }
          
          product.stock -= (Number(item.qty) || 0);
        }
        
        // บันทึกสต็อกสินค้ากลับลง Products sheet
        for (const pid in prodRowMap) {
          const p = prodRowMap[pid];
          prodSheet.getRange(p.rowIndex, 11).setValue(p.stock);
        }
        saveLotsToJson(lots);
        
        // บันทึกรายการใหม่และคำนวณยอดรวมใน transaction
        targetTx.items = checkoutItems;
        const newPartsTotal = checkoutItems.reduce((sum, it) => sum + (it.qty * it.price), 0);
        targetTx.parts_total = newPartsTotal;
        targetTx.total_price = newPartsTotal + (Number(targetTx.labor_cost) || 0);
      }
      
      if (newStatus === "Approved") {
        targetTx.approval_status = "Approved";
        targetTx.approval_date = dateStr;
        targetTx.approval_by = approverName;
        targetTx.approval_by_email = approverEmail;
        targetTx.approval_note = note;
      } else if (newStatus === "Pending") {
        if (note) targetTx.approval_note = note;
      }
    }
    
    saveTransactionsToJson(transactions);
    return {
      status: "success",
      transaction_id: txId,
      approval_status: targetTx.approval_status,
      items: targetTx.items,
      parts_total: targetTx.parts_total,
      total_price: targetTx.total_price,
      message: newStatus === "Approved" ? "อนุมัติเอกสารสำเร็จ" : (newStatus === "Rejected" ? "ปฏิเสธเอกสารและคืนสต็อกสำเร็จ" : "บันทึกการแก้ไขรายการอะไหล่สำเร็จ")
    };
  } finally {
    lock.releaseLock();
  }
}

function deleteTransaction(payload) {

  }
}

function hashPassword(password) {
  const digest = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, password, Utilities.Charset.UTF_8);
  let hexString = "";
  for (let i = 0; i < digest.length; i++) {
    let byteVal = digest[i];
    if (byteVal < 0) byteVal += 256;
    let byteString = byteVal.toString(16);
    if (byteString.length == 1) byteString = "0" + byteString;
    hexString += byteString;
  }
  return hexString;
}

// 12. บันทึกการเพิ่มสต็อกสินค้า (Restock) พร้อมอัปเดตสต็อกในชีต Products และสร้าง Lot ใหม่ลง JSON
function restockProduct(payload) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const prodSheet = ss.getSheetByName("Products");
  
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(30000);
  } catch (e) {
    throw new Error("ระบบไม่ว่างชั่วคราว กรุณาลองใหม่อีกครั้ง");
  }
  
  try {
    const prodValues = prodSheet.getDataRange().getValues();
    const productId = String(payload.id).trim();
    const qtyToAdd = parseFloat(payload.qty) || 0;
    
    if (qtyToAdd === 0) throw new Error("จำนวนที่ปรับเปลี่ยนต้องไม่เป็น 0");
    
    let rowIndex = -1;
    let currentStock = 0;
    let currentCost = 0;
    let currentPriceA = 0, currentPriceB = 0, currentPriceC = 0;
    
    for (let i = 1; i < prodValues.length; i++) {
      if (String(prodValues[i][0]).trim() === productId) {
        rowIndex = i + 1;
        currentCost = parseFloat(prodValues[i][3]) || 0;
        currentPriceA = parseFloat(prodValues[i][4]) || 0;
        currentPriceB = parseFloat(prodValues[i][5]) || 0;
        currentPriceC = parseFloat(prodValues[i][6]) || 0;
        currentStock = parseFloat(prodValues[i][10]) || 0;
        break;
      }
    }
    
    if (rowIndex === -1) throw new Error("ไม่พบอะไหล่รหัส " + productId);
    
    const newStock = currentStock + qtyToAdd;
    if (newStock < 0) throw new Error("ยอดสต็อกหลังปรับปรุงต้องไม่ต่ำกว่า 0 (สต็อกปัจจุบัน: " + currentStock + ", ปรับปรุง: " + qtyToAdd + ")");
    
    // รับค่าต้นทุนและราคาขายใหม่ของ Lot (หากระบุ)
    const lotCost = (payload.cost !== undefined && payload.cost !== "") ? parseFloat(payload.cost) : currentCost;
    const autoPrices = calculatePrices(lotCost);
    const lotPriceA = (payload.price_a !== undefined && payload.price_a !== "") ? parseFloat(payload.price_a) : (currentPriceA || autoPrices.price_a);
    const lotPriceB = (payload.price_b !== undefined && payload.price_b !== "") ? parseFloat(payload.price_b) : (currentPriceB || autoPrices.price_b);
    const lotPriceC = (payload.price_c !== undefined && payload.price_c !== "") ? parseFloat(payload.price_c) : (currentPriceC || autoPrices.price_c);
    
    // อัปเดตสต็อกคงเหลือ และราคาอ้างอิงล่าสุดใน Products Sheet
    prodSheet.getRange(rowIndex, 11).setValue(newStock);
    if (payload.cost !== undefined && payload.cost !== "") {
      prodSheet.getRange(rowIndex, 4, 1, 4).setValues([[lotCost, lotPriceA, lotPriceB, lotPriceC]]);
    }
    
    // สร้าง Lot ใหม่ใน lots.json
    let lots = getLotsFromJson();
    const now = new Date();
    const dateStr = Utilities.formatDate(now, Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss");
    const datePrefix = Utilities.formatDate(now, Session.getScriptTimeZone(), "yyyyMMdd");
    
    const lotId = "LOT-" + datePrefix + "-" + String(Date.now()).slice(-6);
    lots.push({
      lot_id: lotId,
      product_id: productId,
      cost: lotCost,
      price_a: lotPriceA,
      price_b: lotPriceB,
      price_c: lotPriceC,
      initial_qty: qtyToAdd,
      remaining_qty: qtyToAdd,
      created_at: dateStr,
      note: payload.note || "เติมสต็อกอะไหล่"
    });
    saveLotsToJson(lots);
    
    // บันทึกประวัติการเติมสต็อกลงใน transactions.json
    let transactions = getTransactionsFromJson();
    let counter = 1;
    
    if (transactions.length > 0) {
      for (let i = transactions.length - 1; i >= 0; i--) {
        const lastTx = transactions[i];
        if (String(lastTx.id).indexOf("QCM-RE-" + datePrefix) === 0) {
          const parts = String(lastTx.id).split("-");
          const lastNum = parseInt(parts[3], 10);
          if (!isNaN(lastNum)) {
            counter = lastNum + 1;
            break;
          }
        }
      }
    }
    const txId = "QCM-RE-" + datePrefix + "-" + String(counter).padStart(4, '0');
    
    const newRestockTransaction = {
      id: txId,
      date: dateStr,
      requester: payload.requester || "ระบบเติมสต็อก",
      department: payload.department || "สโตร์ (Restock)",
      machine_id: "",
      serial_number: "",
      total_price: qtyToAdd * lotCost,
      note: payload.note || ("เติมสต็อก (Lot: " + lotId + ")"),
      status: "Restock",
      items: [{
        detail_id: txId + "-1",
        product_id: productId,
        lot_id: lotId,
        qty: qtyToAdd,
        unit_cost: lotCost,
        price: lotPriceA,
        subtotal: qtyToAdd * lotCost
      }]
    };
    
    transactions.push(newRestockTransaction);
    saveTransactionsToJson(transactions);
    
    return { new_stock: newStock, transaction_id: txId, lot_id: lotId };
    
  } finally {
    lock.releaseLock();
  }
}

// ฟังก์ชันอัปโหลดไฟล์คู่มือ (PDF หรือ รูปภาพ) ลงใน Google Drive
function uploadManualFileToDrive(base64Data, fileName, mimeType) {
  try {
    if (!base64Data) return "";
    // ถ้าเป็น URL อยู่แล้ว (เช่น ลิงก์เดิม) ไม่ต้องอัพโหลดซ้ำ
    if (base64Data.indexOf('http://') === 0 || base64Data.indexOf('https://') === 0) {
      return base64Data;
    }
    const parts = base64Data.split(',');
    let type = mimeType || "application/pdf";
    if (parts.length > 1 && parts[0].indexOf(';base64') !== -1) {
      const match = parts[0].match(/data:(.*?);base64/);
      if (match && match[1]) type = match[1];
    }
    const base64 = parts[1] || base64Data;
    const decoded = Utilities.base64Decode(base64);
    
    // ตั้งนามสกุลไฟล์ตามประเภท
    let ext = ".pdf";
    if (type.indexOf('image/png') !== -1) ext = ".png";
    else if (type.indexOf('image/jpeg') !== -1 || type.indexOf('image/jpg') !== -1) ext = ".jpg";
    else if (type.indexOf('image/webp') !== -1) ext = ".webp";
    
    const finalFileName = (fileName || ("manual_" + Date.now())) + ext;
    const blob = Utilities.newBlob(decoded, type, finalFileName);
    
    const folder = DriveApp.getFolderById(DRIVE_FOLDER_ID);
    const file = folder.createFile(blob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    
    // ลิงก์ดาวน์โหลดตรงจาก Google Drive
    return 'https://drive.google.com/uc?export=download&id=' + file.getId();
  } catch (e) {
    Logger.log("อัปโหลดไฟล์คู่มือลง Google Drive ล้มเหลว: " + e.toString());
    return base64Data;
  }
}

function addManual(payload) {
  initDatabase();
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = ss.getSheetByName("Manuals");
  const nowStr = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd");
  const id = payload.id || ("MAN-" + String(Date.now()).slice(-6));
  
  let fileUrl = payload.file_url || "";
  if (fileUrl && fileUrl.indexOf("data:") === 0) {
    const uploadedUrl = uploadManualFileToDrive(fileUrl, (payload.title || "manual") + "_" + id, payload.file_type);
    if (uploadedUrl) fileUrl = uploadedUrl;
  }
  
  sheet.appendRow([
    id,
    payload.title || "",
    payload.description || "",
    fileUrl,
    payload.file_type || "",
    nowStr
  ]);
  
  return { id: id, file_url: fileUrl };
}

function editManual(payload) {
  initDatabase();
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = ss.getSheetByName("Manuals");
  const data = sheet.getDataRange().getValues();
  const targetId = String(payload.id).trim();
  
  let fileUrl = payload.file_url || "";
  if (fileUrl && fileUrl.indexOf("data:") === 0) {
    const uploadedUrl = uploadManualFileToDrive(fileUrl, (payload.title || "manual") + "_" + targetId, payload.file_type);
    if (uploadedUrl) fileUrl = uploadedUrl;
  }
  
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]).trim() === targetId) {
      sheet.getRange(i + 1, 2).setValue(payload.title || "");
      sheet.getRange(i + 1, 3).setValue(payload.description || "");
      if (fileUrl) sheet.getRange(i + 1, 4).setValue(fileUrl);
      if (payload.file_type) sheet.getRange(i + 1, 5).setValue(payload.file_type);
      break;
    }
  }
  return { status: "success", file_url: fileUrl };
}

function deleteManual(id) {
  initDatabase();
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = ss.getSheetByName("Manuals");
  const data = sheet.getDataRange().getValues();
  const targetId = String(id).trim();
  
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]).trim() === targetId) {
      sheet.deleteRow(i + 1);
      break;
    }
  }
  return { status: "success" };
}
